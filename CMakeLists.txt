cmake_minimum_required(VERSION 3.20)
project(DaemonCore VERSION 0.1.0 LANGUAGES C CXX)

option(DAEMON_BUILD_TESTS "Build Daemon operational tests" ON)
option(DAEMON_BUILD_CLI "Build the local read-only CLI" ON)
option(DAEMON_PLATFORM_IOS "Disable local processes for an iOS build" OFF)

add_library(daemon_core STATIC
    src/daemon/policy.cpp src/daemon/trace.cpp src/daemon/json.cpp src/daemon/bridge.cpp
    adapters/git_adapter.cpp adapters/posix_process.cpp)
target_compile_features(daemon_core PUBLIC cxx_std_20)
target_include_directories(daemon_core PUBLIC include PRIVATE adapters)
if(DAEMON_PLATFORM_IOS OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(daemon_core PRIVATE DAEMON_PLATFORM_IOS=1)
endif()
if(MSVC)
    target_compile_options(daemon_core PRIVATE /W4)
else()
    target_compile_options(daemon_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(DAEMON_BUILD_CLI)
    add_executable(daemon_cli cli/main.cpp)
    target_link_libraries(daemon_cli PRIVATE daemon_core)
    set_target_properties(daemon_cli PROPERTIES OUTPUT_NAME daemon)
endif()

if(DAEMON_BUILD_TESTS)
    include(CTest)
    find_package(Threads REQUIRED)
    add_executable(daemon_tests tests/daemon/tests.cpp)
    target_link_libraries(daemon_tests PRIVATE daemon_core Threads::Threads)
    target_include_directories(daemon_tests PRIVATE adapters)
    target_compile_definitions(daemon_tests PRIVATE DAEMON_TEST_REPOSITORY="${CMAKE_CURRENT_SOURCE_DIR}")
    if(DAEMON_PLATFORM_IOS OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        target_compile_definitions(daemon_tests PRIVATE DAEMON_PLATFORM_IOS=1)
    endif()
    add_test(NAME daemon_unit COMMAND daemon_tests)
    if(DAEMON_BUILD_CLI AND NOT DAEMON_PLATFORM_IOS AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
        add_test(NAME daemon_cli_snapshot COMMAND daemon_cli snapshot --repo ${CMAKE_CURRENT_SOURCE_DIR} --json)
        add_test(NAME daemon_cli_explicit_path COMMAND daemon_cli snapshot)
        set_tests_properties(daemon_cli_explicit_path PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

install(TARGETS daemon_core ARCHIVE DESTINATION lib)
install(FILES include/daemon/daemon_bridge.h DESTINATION include/daemon)
if(DAEMON_BUILD_CLI)
    install(TARGETS daemon_cli RUNTIME DESTINATION bin)
endif()
